import { useScroll } from "motion/react";
import { ZoomIn, ZoomOut } from "lucide-react";
import { useEffect, useRef, useState } from "react";
import { useReached } from "@/lib/motion-hooks";
import { useMotionOk } from "@/lib/use-motion-ok";
import { cn } from "@/lib/utils";
import type { FabricScene } from "./knit/fabric-scene";

/** A draped swatch of fine cotton single jersey (scripts/build-knit-glb.mjs). */
const MODEL = "/models/cotton-jersey-v4.glb";
const POSTER = "/images/knit-lattice-poster.webp";

/**
 * The knit plate: fine cotton single jersey lying in soft folds, lit like a textile sample.
 * Scrolling past moves the view from the drape in towards the knit and back; the pointer adds a
 * gentle parallax and a warm light that rakes across the fibres; Inspect stitches brings the
 * knit up close, where the pointer moves the close-up across the cloth.
 *
 * three.js and the scene load only when the plate is near the viewport; frames render only
 * while something changes. The poster is the default composition: shown before load, without
 * WebGL and in the server render. Reduced motion: no scroll movement or parallax, and the
 * close-up switches without a transition.
 */
export function KnitModel({ className }: { className?: string }) {
  const plate = useRef<HTMLDivElement>(null);
  const host = useRef<HTMLDivElement>(null);
  const motionOk = useMotionOk();
  const motionRef = useRef(motionOk);
  motionRef.current = motionOk;
  const near = useReached(plate, 1.8);
  const { scrollYProgress } = useScroll({ target: plate, offset: ["start end", "end start"] });
  const sceneRef = useRef<FabricScene | null>(null);
  const [status, setStatus] = useState<"waiting" | "live" | "failed">("waiting");
  const [inspect, setInspect] = useState(false);
  const inspectRef = useRef(inspect);
  inspectRef.current = inspect;

  useEffect(() => {
    const frame = plate.current;
    const mount = host.current;
    if (!near || !frame || !mount) return;
    let disposed = false;
    const memory = (navigator as Navigator & { deviceMemory?: number }).deviceMemory ?? 8;
    const lite =
      window.matchMedia("(max-width: 767px)").matches || memory <= 4 || (navigator.hardwareConcurrency ?? 8) <= 2;

    import("./knit/fabric-scene")
      .then(({ createFabricScene }) =>
        createFabricScene({ host: mount, frame, lite, motionOk: () => motionRef.current, modelUrl: MODEL }),
      )
      .then((scene) => {
        if (disposed) return scene.dispose();
        sceneRef.current = scene;
        scene.setScroll(scrollYProgress.get());
        scene.setInspect(inspectRef.current);
        setStatus("live");
      })
      .catch((error: unknown) => {
        console.warn("Fabric model unavailable; showing the poster.", error);
        if (!disposed) setStatus("failed");
      });

    const unsubscribe = scrollYProgress.on("change", (value) => sceneRef.current?.setScroll(value));
    return () => {
      disposed = true;
      unsubscribe();
      sceneRef.current?.dispose();
      sceneRef.current = null;
    };
  }, [near, scrollYProgress]);

  useEffect(() => {
    sceneRef.current?.setInspect(inspect);
    if (!inspect) return;
    const onKey = (event: KeyboardEvent) => {
      if (event.key === "Escape") setInspect(false);
    };
    window.addEventListener("keydown", onKey);
    return () => window.removeEventListener("keydown", onKey);
  }, [inspect]);

  const live = status === "live";

  return (
    <div className={className}>
      <div
        ref={plate}
        role="img"
        aria-label="Fine cotton single jersey lying in soft folds: an illustrative 3D render of the knit."
        className="grain relative aspect-[4/3] select-none overflow-hidden bg-[radial-gradient(120%_95%_at_42%_36%,var(--charcoal-2),var(--charcoal)_72%)]"
      >
        <img
          src={POSTER}
          alt=""
          aria-hidden
          loading="lazy"
          decoding="async"
          className={cn(
            "absolute inset-0 h-full w-full object-cover transition-opacity duration-700 ease-[var(--ease-out)]",
            live && "opacity-0",
          )}
        />
        <div
          ref={host}
          className={cn(
            "absolute inset-0 opacity-0 transition-opacity duration-700 ease-[var(--ease-out)]",
            live && "opacity-100",
          )}
        />
      </div>

      <div className="mt-4 flex flex-wrap items-center justify-between gap-x-6 gap-y-2">
        <p className="text-sm text-ink-soft">
          Fine cotton single jersey, an illustrative render.
          {live && (
            <span className="hidden [@media(hover:hover)_and_(pointer:fine)]:inline">
              {" "}
              {inspect ? "Move across the cloth to explore the stitches." : "Move across the fabric to raise the light."}
            </span>
          )}
        </p>
        {status !== "failed" && (
          <button
            type="button"
            disabled={!live}
            aria-pressed={inspect}
            onClick={() => setInspect((value) => !value)}
            className="inline-flex min-h-11 items-center gap-2 px-3 text-sm font-medium text-ink transition-colors duration-200 hover:bg-linen disabled:pointer-events-none disabled:opacity-40"
          >
            {inspect ? <ZoomOut aria-hidden className="size-4" /> : <ZoomIn aria-hidden className="size-4" />}
            {inspect ? "Full view" : "Inspect stitches"}
          </button>
        )}
      </div>
    </div>
  );
}
