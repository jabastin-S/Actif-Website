import { useScroll } from "motion/react";
import type { LineSegments, Material, Mesh, Object3D, WebGLProgramParametersWithUniforms, WebGLRenderer } from "three";
import { useEffect, useRef, useState } from "react";
import { useReached } from "@/lib/motion-hooks";
import { useMotionOk } from "@/lib/use-motion-ok";
import { cn } from "@/lib/utils";

/** Desktop and phone builds of the same swatch (scripts/build-knit-glb.mjs). */
const MODEL = "/models/knit-lattice-v2.glb";
const MODEL_LITE = "/models/knit-lattice-v2-lite.glb";
const POSTER = "/images/knit-lattice-poster.webp";

const clamp01 = (v: number) => Math.min(1, Math.max(0, v));
const lerp = (a: number, b: number, t: number) => a + (b - a) * t;
const ease = (t: number) => 1 - Math.pow(1 - t, 3);

/** Any CSS colour (the plate uses oklch tokens) as an sRGB hex, via a 1x1 canvas. */
function cssToHex(color: string) {
  const context = document.createElement("canvas").getContext("2d");
  if (!context) return "#1c1b18";
  context.fillStyle = color;
  context.fillRect(0, 0, 1, 1);
  const [r = 0, g = 0, b = 0] = context.getImageData(0, 0, 1, 1).data;
  return `#${[r, g, b].map((v) => v.toString(16).padStart(2, "0")).join("")}`;
}

/** How much of course `i` is knitted at scroll progress `p`. */
const knitted = (p: number, i: number) => clamp01((p - i * 0.15) / 0.4);

/**
 * Discards fragments beyond the knitted share of the yarn. The share is read from uv.x (0..1
 * along the ply, written by the generator), not from index order, so the compressed, reordered
 * file still knits from one end of each course to the other.
 */
function knitReveal(material: Material, share: { value: number }) {
  material.onBeforeCompile = (shader: WebGLProgramParametersWithUniforms) => {
    shader.uniforms["uKnit"] = share;
    shader.vertexShader = shader.vertexShader
      .replace("#include <common>", "#include <common>\nvarying float vKnit;")
      .replace("#include <begin_vertex>", "#include <begin_vertex>\nvKnit = uv.x;");
    shader.fragmentShader = shader.fragmentShader
      .replace("#include <common>", "#include <common>\nuniform float uKnit;\nvarying float vKnit;")
      .replace("#include <clipping_planes_fragment>", "if (vKnit > uKnit) discard;\n#include <clipping_planes_fragment>");
  };
  material.customProgramCacheKey = () => "knit-reveal";
}

/**
 * The knit swatch on a charcoal plate. As the plate rises to the middle of the screen each
 * course is knitted in turn while the camera eases in from a wider view to a macro crop, the
 * swatch turns to face the reader and the key light rakes across the twist. On a fine pointer
 * it tilts gently towards the cursor.
 *
 * Look: three twisted plies per yarn with a fibre normal map, baked contact occlusion and
 * restrained fuzz (in the asset); a studio environment for soft image-based light, a
 * shadow-casting key, a cool rim and depth fog matched to the plate.
 *
 * three.js is imported only when the plate is near, frames render on demand (scroll, pointer,
 * resize; nothing while idle), and phones or low-core devices get the lighter build. The poster
 * is the finished swatch: shown before load, without WebGL and in the server render. Reduced
 * motion shows the finished swatch, still.
 */
export function KnitModel({ className }: { className?: string }) {
  const plate = useRef<HTMLDivElement>(null);
  const host = useRef<HTMLDivElement>(null);
  const motionOk = useMotionOk();
  const motionRef = useRef(motionOk);
  motionRef.current = motionOk;
  const near = useReached(plate, 1.8);
  const { scrollYProgress } = useScroll({ target: plate, offset: ["start end", "center center"] });
  const [live, setLive] = useState(false);

  useEffect(() => {
    const mount = host.current;
    const frame = plate.current;
    if (!near || !mount || !frame) return;
    let disposed = false;
    const cleanups: (() => void)[] = [];
    // ?knit=lite|full forces a build for QA; otherwise phones and constrained devices (<= 4 GB,
    // <= 2 cores) get the lite one. A 4-thread, 8 GB desktop holds 60 fps on the full build.
    const forced = new URLSearchParams(window.location.search).get("knit");
    const memory = (navigator as Navigator & { deviceMemory?: number }).deviceMemory ?? 8;
    const constrained =
      window.matchMedia("(max-width: 767px)").matches || memory <= 4 || (navigator.hardwareConcurrency ?? 8) <= 2;
    const lite = forced === "lite" || (forced !== "full" && constrained);

    (async () => {
      const [THREE, { GLTFLoader }, { MeshoptDecoder }, { RoomEnvironment }] = await Promise.all([
        import("three"),
        import("three/examples/jsm/loaders/GLTFLoader.js"),
        import("three/examples/jsm/libs/meshopt_decoder.module.js"),
        import("three/examples/jsm/environments/RoomEnvironment.js"),
      ]);
      if (disposed) return;

      let renderer: WebGLRenderer;
      try {
        renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true, powerPreference: "low-power" });
      } catch {
        return; // no WebGL: the poster stays
      }
      renderer.setPixelRatio(Math.min(window.devicePixelRatio, lite ? 1.5 : 2));
      // Khronos PBR Neutral keeps the cotton's hue (AgX greyed it)
      renderer.toneMapping = THREE.NeutralToneMapping;
      renderer.toneMappingExposure = 1;
      renderer.shadowMap.enabled = true;
      renderer.shadowMap.type = THREE.PCFShadowMap;
      renderer.domElement.setAttribute("aria-hidden", "true");
      renderer.domElement.className = "block h-full w-full";

      const scene = new THREE.Scene();
      const camera = new THREE.PerspectiveCamera(24, 1, 0.1, 100);

      // soft studio light for natural highlight roll-off, plus a raking key, cool rim and low fill
      const pmrem = new THREE.PMREMGenerator(renderer);
      const environment = pmrem.fromScene(new RoomEnvironment(), 0.04).texture;
      pmrem.dispose();
      scene.environment = environment;
      scene.environmentIntensity = 0.26;
      // the key is a wide, soft-edged spot: light pools across the centre of the swatch and falls
      // away towards the corners, as in textile product photography
      const key = new THREE.SpotLight(0xffeed6, 3.4, 0, 0.42, 0.95, 0);
      key.castShadow = true;
      key.shadow.mapSize.setScalar(lite ? 1024 : 2048);
      key.shadow.bias = -0.0004;
      key.shadow.normalBias = 0.012;
      key.shadow.radius = 3;
      key.shadow.camera.near = 4;
      key.shadow.camera.far = 24;
      const rim = new THREE.DirectionalLight(0xc6d6e8, 1.25);
      rim.position.set(6, 3, -7);
      scene.add(key, key.target, rim, new THREE.HemisphereLight(0xfff4e6, 0x17150f, 0.16));

      // depth: the back of the swatch falls away into the plate's own charcoal
      const ground = new THREE.Color(cssToHex(getComputedStyle(frame).backgroundColor));
      scene.fog = new THREE.Fog(ground, 1, 2);

      const loader = new GLTFLoader().setMeshoptDecoder(MeshoptDecoder);
      const gltf = await loader.loadAsync(lite ? MODEL_LITE : MODEL);
      if (disposed) {
        environment.dispose();
        renderer.dispose();
        return;
      }
      const swatch = gltf.scene;
      scene.add(swatch);

      // per course: a share uniform driving its yarn, its fuzz and its shadow
      const courses = swatch.children[0]?.children.length ? swatch.children[0].children : swatch.children;
      const owned: Material[] = [];
      const shares = [...courses]
        .sort((a, b) => a.name.localeCompare(b.name, undefined, { numeric: true }))
        .map((course) => {
          const share = { value: 1 };
          const depth = new THREE.MeshDepthMaterial({ depthPacking: THREE.RGBADepthPacking });
          knitReveal(depth, share);
          owned.push(depth);
          course.traverse((object: Object3D) => {
            const part = object as Mesh | LineSegments;
            if (!("isMesh" in part && part.isMesh) && !("isLineSegments" in part && part.isLineSegments)) return;
            const material = (part.material as Material).clone();
            knitReveal(material, share);
            owned.push(material);
            part.material = material;
            if ("isMesh" in part && part.isMesh) {
              part.castShadow = true;
              part.receiveShadow = true;
              part.customDepthMaterial = depth;
            }
          });
          return share;
        });
      mount.appendChild(renderer.domElement);

      // framing: fit the swatch to the plate, then ease from a wider view into a macro crop
      const size = new THREE.Box3().setFromObject(swatch).getSize(new THREE.Vector3());
      let fitDistance = 10;
      const fit = () => {
        const { width, height } = frame.getBoundingClientRect();
        if (!width || !height) return;
        renderer.setSize(width, height, false);
        camera.aspect = width / height;
        const halfV = THREE.MathUtils.degToRad(camera.fov / 2);
        fitDistance = Math.min((size.y * 0.5) / Math.tan(halfV), (size.x * 0.5) / (Math.tan(halfV) * camera.aspect));
        camera.updateProjectionMatrix();
      };

      let pending = 0;
      const render = () => {
        if (!pending) {
          pending = requestAnimationFrame(() => {
            pending = 0;
            renderer.render(scene, camera);
          });
        }
      };

      const tilt = { x: 0, y: 0, tx: 0, ty: 0 };
      let progress = 1;
      const apply = () => {
        const p = motionRef.current ? progress : 1;
        shares.forEach((share, i) => (share.value = knitted(p, i)));
        const turn = ease(p);
        swatch.rotation.set(lerp(0.78, 0.36, turn) + tilt.y, lerp(-0.85, -0.42, turn) + tilt.x, lerp(0.14, 0.06, turn));
        const distance = fitDistance * lerp(1.12, 0.8, turn);
        camera.position.set(0, lerp(0.25, 0, turn), distance);
        camera.lookAt(0, 0, 0);
        const fog = scene.fog as InstanceType<typeof THREE.Fog>;
        fog.near = distance - 0.2;
        fog.far = distance + 7;
        // the key rakes across the yarn as the swatch settles, revealing the twist
        const azimuth = THREE.MathUtils.degToRad(lerp(-62, -38, turn));
        const elevation = THREE.MathUtils.degToRad(lerp(48, 36, turn));
        key.position.set(Math.sin(azimuth) * Math.cos(elevation) * 12, Math.sin(elevation) * 12, Math.cos(azimuth) * Math.cos(elevation) * 12);
        render();
      };

      progress = scrollYProgress.get();
      cleanups.push(
        scrollYProgress.on("change", (value) => {
          progress = value;
          apply();
        }),
      );

      // pointer: ease towards the cursor, then stop rendering once settled
      let easing = 0;
      const step = () => {
        tilt.x += (tilt.tx - tilt.x) * 0.12;
        tilt.y += (tilt.ty - tilt.y) * 0.12;
        apply();
        easing = Math.abs(tilt.tx - tilt.x) + Math.abs(tilt.ty - tilt.y) > 0.0005 ? requestAnimationFrame(step) : 0;
      };
      const aim = (event: PointerEvent) => {
        if (event.pointerType !== "mouse" || !motionRef.current) return;
        const box = frame.getBoundingClientRect();
        tilt.tx = ((event.clientX - box.left) / box.width - 0.5) * 0.24;
        tilt.ty = ((event.clientY - box.top) / box.height - 0.5) * 0.18;
        if (!easing) easing = requestAnimationFrame(step);
      };
      const release = () => {
        tilt.tx = 0;
        tilt.ty = 0;
        if (!easing) easing = requestAnimationFrame(step);
      };
      frame.addEventListener("pointermove", aim);
      frame.addEventListener("pointerleave", release);

      const resize = new ResizeObserver(() => {
        fit();
        apply();
      });
      resize.observe(frame);

      cleanups.push(() => {
        frame.removeEventListener("pointermove", aim);
        frame.removeEventListener("pointerleave", release);
        resize.disconnect();
        cancelAnimationFrame(pending);
        cancelAnimationFrame(easing);
        swatch.traverse((object: Object3D) => {
          const part = object as Mesh;
          if (!part.geometry) return;
          part.geometry.dispose();
        });
        const textures = new Set<{ dispose: () => void }>();
        owned.forEach((material) => {
          for (const value of Object.values(material)) if (value instanceof THREE.Texture) textures.add(value);
          material.dispose();
        });
        textures.forEach((texture) => texture.dispose());
        environment.dispose();
        renderer.dispose();
        renderer.domElement.remove();
      });

      fit();
      apply();
      setLive(true);
    })().catch((error: unknown) => {
      // a failed load leaves the poster in place
      console.warn("Knit model unavailable; showing the poster.", error);
    });

    return () => {
      disposed = true;
      cleanups.forEach((fn) => fn());
    };
  }, [near, scrollYProgress]);

  return (
    <div ref={plate} className={cn("grain relative overflow-hidden bg-charcoal", className)}>
      <img
        src={POSTER}
        alt=""
        aria-hidden
        loading="lazy"
        decoding="async"
        className={cn(
          "absolute inset-0 h-full w-full object-cover transition-opacity duration-700 ease-[var(--ease-out)]",
          live && "opacity-0",
        )}
      />
      <div
        ref={host}
        className={cn(
          "absolute inset-0 opacity-0 transition-opacity duration-700 ease-[var(--ease-out)]",
          live && "opacity-100",
        )}
      />
      <span className="sr-only">
        A model of single jersey knitting: courses of plied cotton yarn, each loop drawn through
        the course below.
      </span>
    </div>
  );
}
